\setcounter{enumi}{58}
\item {\bfseries\boldmath
Determine whether each of the following sets, equipped with the indicated operations, is a real vector space.
}

\begin{enumerate}[label=\alph*)]

\item $\RR^+$ with binary operation $\oplus$ and scalar multiplication $\otimes$ defined by
\[
x\oplus y=xy,\qquad\alpha\otimes x=x^\alpha
\]
for all $x,y\in\RR^+$ and $\alpha\in\RR$.

\begin{answer}
Yes. On $\RR^+$, $x\oplus y=xy$ is closed, associative and commutative.
Its neutral element is $1$, and the additive inverse of $x$ for this operation
is $1/x$. Scalar multiplication is $\alpha\otimes x=x^\alpha>0$. For
$x,y>0$ and $\alpha,\beta\in\RR$,
\[
\begin{aligned}
 1\otimes x&=x,\\
 \alpha\otimes(\beta\otimes x)&=(x^\beta)^\alpha=x^{\alpha\beta},\\
 \alpha\otimes(x\oplus y)&=(xy)^\alpha=x^\alpha y^\alpha
   =(\alpha\otimes x)\oplus(\alpha\otimes y),\\
 (\alpha+\beta)\otimes x&=x^{\alpha+\beta}=x^\alpha x^\beta
   =(\alpha\otimes x)\oplus(\beta\otimes x).
\end{aligned}
\]
These are the remaining vector space axioms. The zero vector of this space
is the real number $1$.
\end{answer}

\item $\RR$ with its usual addition and scalar multiplication $\otimes$ defined by
\[
\alpha\otimes x=\alpha^2x
\]
for all $x,\alpha\in\RR$.

\begin{answer}
No. With $\alpha\otimes x=\alpha^2x$,
\[
 (1+1)\otimes1=4\ne2=(1\otimes1)+(1\otimes1).
\]
Distributivity with respect to scalar addition fails.
\end{answer}

\item $\RR^2$ with its usual addition and scalar multiplication $\otimes$ defined by
\[
\alpha\otimes(x,y)=(\alpha x,0)
\]
for all $x,y,\alpha\in\RR$.

\begin{answer}
No. For $v=(0,1)$, $1\otimes v=(0,0)\ne v$, violating the scalar identity axiom.
\end{answer}

\item $\RR^2$ with its usual addition and scalar multiplication $\otimes$ defined by
\[
\alpha\otimes(x,y)=(\alpha x,y)
\]
for all $x,y,\alpha\in\RR$.

\begin{answer}
No. For $v=(0,1)$,
$(1+1)\otimes v=(0,1)$, whereas
$(1\otimes v)+(1\otimes v)=(0,2)$.
\end{answer}

\item The set $\RR^{\mathbb N}$ of all real sequences with the usual addition and scalar multiplication:
\[
\begin{gathered}(x_n)_{n\in\mathbb N}+(y_n)_{n\in\mathbb N}=(x_n+y_n)_{n\in\mathbb N},\\\alpha(x_n)_{n\in\mathbb N}=(\alpha x_n)_{n\in\mathbb N},\end{gathered}
\]
for all real sequences $(x_n)$, $(y_n)$ and every $\alpha\in\RR$.

\begin{answer}
Yes. Addition and scalar multiplication of real sequences are defined
coordinatewise. All real vector space identities hold in each coordinate,
and therefore hold for the sequences. The zero vector is $(0)_{n\in\mathbb N}$;
the additive inverse of $(x_n)$ is $(-x_n)$.
\end{answer}
\end{enumerate}
