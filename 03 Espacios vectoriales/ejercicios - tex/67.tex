% ---------- Exercise 67 ----------
\setcounter{enumi}{66}
\item {\bfseries\boldmath
State, with justification, whether in the real vector space $\RR[x]$ the polynomial $-12x^3-5x^2+2x-4$ is a linear combination of
}

\begin{enumerate}[label=\alph*)]

\item $\bigl(2x^3+3x-1,\ -x^2+4x-2\bigr)$.

\begin{answer}
Let $p=-12x^3-5x^2+2x-4$. Polynomial equality is checked coefficientwise.

Yes:
\[
 p=-6(2x^3+3x-1)+5(-x^2+4x-2).
\]
\end{answer}

\item $\bigl(-2x^3-x^2+x-3,\ 2x^3+\frac12x^2+x-4\bigr)$.

\begin{answer}
Yes:
\[
 p=4(-2x^3-x^2+x-3)-2(2x^3+\tfrac12x^2+x-4).
\]
\end{answer}

\item $\bigl(-2x^3-x^2+x-1,\ 2x^3+x^2+2x+2,\ -4x^3-x^2+4x-2\bigr)$.

\begin{answer}
No. For a polynomial $ax^3+bx^2+cx+d$, each of the three given
generators satisfies $-3a+2b-c+3d=0$.
This coefficient identity is preserved by linear combinations, but for $p$
it gives $36-10-2-12=12\ne0$.
\end{answer}

\item
\[
\bigl(2x^3-x^2+2x-2,\ -3x^3-4x^2+4x-5,
\]

\[
\hspace{1em}-7x^3-2x^2-9,\ -x^3-5x^2+6x-7\bigr).
\]

\begin{answer}
Yes. The first two generators already suffice:
\[
 p=-3(2x^3-x^2+2x-2)+2(-3x^3-4x^2+4x-5).
\]
The coefficients of the third and fourth generators can both be zero.
\end{answer}
\end{enumerate}
