\setcounter{enumi}{69}
\item {\bfseries\boldmath
Let $F=\langle(0,1,3,1),(1,0,0,-1)\rangle$ in the real vector space $\RR^4$. Give four elements of $F$.
}

\begin{answer}
For $u=(0,1,3,1)$ and $v=(1,0,0,-1)$, the space is $F=\langle u,v\rangle$.
Four distinct elements are
\[
 0=(0,0,0,0),\qquad u=(0,1,3,1),\qquad v=(1,0,0,-1),\qquad
 u+v=(1,1,3,0).
\]
\end{answer}
