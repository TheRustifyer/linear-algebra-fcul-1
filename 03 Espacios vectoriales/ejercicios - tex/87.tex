\setcounter{enumi}{86}
\item {\bfseries\boldmath
Give a basis and the dimension of
}

\begin{enumerate}[label=\alph*)]

\item $\CC$ as a complex vector space.

\begin{answer}
As a complex vector space, $\CC$ has basis $(1)$ and dimension $1$:
every $z\in\CC$ is the complex scalar multiple $z\cdot1$.
\end{answer}

\item $\CC$ as a real vector space.

\begin{answer}
As a real vector space, $\CC$ has basis $(1,i)$ and dimension $2$.
Every complex number has a unique expression $a+bi$, $a,b\in\RR$.
\end{answer}

\item $\CC^3$ as a complex vector space.

\begin{answer}
As a complex vector space, $\CC^3$ has basis
\[
 (e_1,e_2,e_3),\qquad e_1=(1,0,0),\ e_2=(0,1,0),\ e_3=(0,0,1),
\]
and dimension $3$, because the three complex coordinates are arbitrary and unique.
\end{answer}

\item $\CC^3$ as a real vector space.

\begin{answer}
As a real vector space, $\CC^3$ has basis
$(e_1,ie_1,e_2,ie_2,e_3,ie_3)$ and dimension $6$.
Indeed, each complex coordinate has unique real and imaginary parts.
\end{answer}
\end{enumerate}
