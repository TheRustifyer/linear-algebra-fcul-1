% ---------- Exercise 75 ----------
\setcounter{enumi}{74}
\item {\bfseries\boldmath
Consider the real vector space $\RR^3$. Determine the real values of $t,k$ for which
}

\[
(t,k,-t)\in\langle(1,t,1),(1,2t,-2),(-1,-t,t)\rangle.
\]

\begin{answer}
Seek coefficients $p,q,r$ for the generators in the order given:
\[
 p(1,t,1)+q(1,2t,-2)+r(-1,-t,t)=(t,k,-t).
\]
The augmented matrix, after $F_2\gets F_2-tF_1$ and $F_3\gets F_3-F_1$, is
\[
 \left[\begin{array}{rrr|r}
 1&1&-1&t\\0&t&0&k-t^2\\0&-3&t+1&-2t
 \end{array}\right].
\]
If $t\notin\{0,-1\}$, there are three pivots, so every real $k$ is allowed.
If $t=0$, row $2$ requires $k=0$; the target is then the zero vector and is
in the span.
If $t=-1$, row $3$ gives $q=-2/3$, and row $2$ gives $q=1-k$.
Consequently $k=5/3$. Sufficiency follows from
\[
 -\tfrac13(1,-1,1)-\tfrac23(1,-2,-2)=(-1,5/3,1).
\]
The full answer for the displayed enunciation is therefore
\[
 \boxed{\begin{gathered}
 t\notin\{0,-1\},\ k\in\RR;\qquad t=0,\ k=0;\\
 t=-1,\ k=5/3.
 \end{gathered}}
\]
The published answer gives $k=1/3$ in the last case. That value contradicts
the coordinate equations above; $5/3$ is the value for the vectors printed
in the exercise sheet.
\end{answer}
