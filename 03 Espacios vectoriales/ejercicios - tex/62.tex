\setcounter{enumi}{61}
\item {\bfseries\boldmath
Consider the real vector space $M_3(\RR)$. State, with justification, whether each of the following sets is a vector subspace of $M_3(\RR)$.
}

\begin{enumerate}[label=\alph*)]

\item $W_1=\left\{\begin{bmatrix}0&x&y\\-x&2y&0\\2x&0&-y\end{bmatrix}:x,y\in\RR\right\}$.

\begin{answer}
Yes. Write
\[
 \begin{bmatrix}0&x&y\\-x&2y&0\\2x&0&-y\end{bmatrix}
 =x\begin{bmatrix}0&1&0\\-1&0&0\\2&0&0\end{bmatrix}
 +y\begin{bmatrix}0&0&1\\0&2&0\\0&0&-1\end{bmatrix}.
\]
The set is a span, so it is a subspace.
\end{answer}

\item $W_2=\left\{\begin{bmatrix}0&x&y\\-x&2y&1\\2x&0&-y\end{bmatrix}:x,y\in\RR\right\}$.

\begin{answer}
No. Every matrix in $W_2$ has entry $(2,3)$ equal to $1$, so the zero matrix is absent.
\end{answer}

\item $W_3=\{[a_{ij}]\in M_3(\RR):a_{11}+a_{23}=0,\ a_{21}-5a_{33}=0\}$.

\begin{answer}
Yes. The conditions $a_{11}+a_{23}=0$ and $a_{21}-5a_{33}=0$ are
homogeneous linear equations in the entries. The zero matrix satisfies them,
and any linear combination of two matrices satisfying them does too.
\end{answer}

\item $W_4=\{[a_{ij}]\in M_3(\RR):a_{11}=a_{12}=a_{13}\}$.

\begin{answer}
Yes. Rewrite the conditions as $a_{11}-a_{12}=0$ and
$a_{11}-a_{13}=0$, and apply the same criterion.
\end{answer}

\item $W_5=\{[a_{ij}]\in M_3(\RR):a_{21}\ge a_{23}\}$.

\begin{answer}
No. The matrix with sole nonzero entry $a_{21}=1$ satisfies
$a_{21}\ge a_{23}$. Its negative does not.
\end{answer}
\end{enumerate}
