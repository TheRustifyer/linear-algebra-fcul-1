% ---------- Exercise 61 ----------
\setcounter{enumi}{60}
\item {\bfseries\boldmath
Consider the real vector space $\RR^3$. State, with justification, whether each of the following sets is a vector subspace of $\RR^3$.
}

\begin{enumerate}[label=\alph*)]

\item $V_1=\{(x,y,z)\in\RR^3:x-y+z=0\}$.

\begin{answer}
A subset containing $0$ is a subspace if it is closed under linear
combinations $\alpha u+\beta v$. In particular, homogeneous linear conditions
are preserved by such combinations.

$V_1=\{(x,y,z):x-y+z=0\}$ is a subspace: its defining equation is homogeneous and linear.
\end{answer}

\item $V_2=\{(x,y,z)\in\RR^3:x-y+z=0,\ x+y-2z=0\}$.

\begin{answer}
$V_2=\{(x,y,z):x-y+z=0,\ x+y-2z=0\}$ is a subspace for the same reason.
\end{answer}

\item $V_3=\{(x,y,z)\in\RR^3:x-y+z=0,\ x+y-2z=1\}$.

\begin{answer}
$V_3$ is not a subspace, since $0$ does not satisfy $x+y-2z=1$.
\end{answer}

\item $V_4=\{(x,y,z)\in\RR^3:x\text{ is rational}\}$.

\begin{answer}
$V_4$ is not a subspace. It contains $(1,0,0)$ but not
$\sqrt2(1,0,0)$, whose first coordinate is irrational.
\end{answer}

\item $V_5=\{(x,y,z)\in\RR^3:|x|=|y|\}$.

\begin{answer}
$V_5$ is not a subspace. Both $(1,1,0)$ and $(1,-1,0)$ satisfy
$|x|=|y|$, but their sum $(2,0,0)$ does not.
\end{answer}

\item $V_6=\{(x,y,z)\in\RR^3:xy\ge0\}$.

\begin{answer}
$V_6$ is not a subspace. The vectors $(1,0,0)$ and $(0,-1,0)$ satisfy
$xy\ge0$, but their sum $(1,-1,0)$ does not.
\end{answer}

\item $V_7=\{(0,a,b):a,b\in\RR\}$.

\begin{answer}
$V_7=\{(0,a,b):a,b\in\RR\}$ is a subspace: linear combinations still
have first coordinate zero.
\end{answer}

\item $V_8=\{(1,a,b):a,b\in\RR\}$.

\begin{answer}
$V_8=\{(1,a,b):a,b\in\RR\}$ is not a subspace, since it does not contain $0$.
\end{answer}

\item $V_9=\{(0,0,0),(0,1,0),(0,-1,0)\}$.

\begin{answer}
$V_9$ is not a subspace. It contains $(0,1,0)$ but not twice this vector.
\end{answer}

\item $V_{10}=\{(2a,b,-3a):a,b\in\RR\}$.

\begin{answer}
$V_{10}=\{(2a,b,-3a):a,b\in\RR\}$ is a subspace: the zero vector occurs
for $a=b=0$, and linear combinations correspond to taking the same linear
combinations of the parameters $a,b$.
\end{answer}

\item $V_{11}=\{(x,y,z)\in\RR^3:x+2y=0\ \land\ z=0\}$.

\begin{answer}
$V_{11}=\{(x,y,z):x+2y=0,\ z=0\}$ is a subspace, being defined by
homogeneous linear equations.
\end{answer}

\item $V_{12}=\{(x,y,z)\in\RR^3:x+2y=0\ \lor\ z=0\}$.

\begin{answer}
$V_{12}=\{(x,y,z):x+2y=0\text{ or }z=0\}$ is not a subspace.
It contains $(0,0,1)$ and $(1,0,0)$, but their sum $(1,0,1)$ satisfies neither condition.
\end{answer}
\end{enumerate}
