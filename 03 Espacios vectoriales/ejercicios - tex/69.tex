\setcounter{enumi}{68}
\item {\bfseries\boldmath
State, with justification, whether in the real vector space $\mathcal F(\RR)$
}

\begin{enumerate}[label=\alph*)]

\item the function $\cos(2x)$ is a linear combination of $\bigl(\sin^2x,\cos^2x\bigr)$.

\begin{answer}
Yes: $\cos(2x)=-\sin^2x+\cos^2x$ for every $x\in\RR$.
\end{answer}

\item the function $\cos x$ is a linear combination of $\bigl(\sin^2x,\cos^2x\bigr)$.

\begin{answer}
No. If $\cos x=\alpha\sin^2x+\beta\cos^2x$ for all $x$, evaluation
at $x=0$ gives $\beta=1$, whereas evaluation at $x=\pi$ gives $\beta=-1$.
\end{answer}
\end{enumerate}
