\setcounter{enumi}{71}
\item {\bfseries\boldmath
Give a geometric interpretation of the vector subspace
}

\begin{enumerate}[label=\alph*)]

\item $\langle(2,-3)\rangle$ of $\RR^2$.

\begin{answer}
$\langle(2,-3)\rangle=\{(2t,-3t):t\in\RR\}$ is the line through the
origin with direction $(2,-3)$, equivalently $3x+2y=0$.
\end{answer}

\item $\langle(2,-3),(-4,6)\rangle$ of $\RR^2$.

\begin{answer}
Since $(-4,6)=-2(2,-3)$, the span is the same line as in (a).
\end{answer}

\item $\langle(2,-3,1)\rangle$ of $\RR^3$.

\begin{answer}
$\langle(2,-3,1)\rangle=\{(2t,-3t,t):t\in\RR\}$ is a line through
the origin in $\RR^3$, with direction $(2,-3,1)$.
\end{answer}

\item $\langle(1,0,0),(0,1,0)\rangle$ of $\RR^3$.

\begin{answer}
$\langle(1,0,0),(0,1,0)\rangle=\{(x,y,0):x,y\in\RR\}$ is the
coordinate plane $z=0$.
\end{answer}
\end{enumerate}
