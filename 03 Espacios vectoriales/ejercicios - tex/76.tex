% ---------- Exercise 76 ----------
\setcounter{enumi}{75}
\item {\bfseries\boldmath
Consider the vector subspace of $\RR^4$
}

\[
V=\langle(2,2,4,-2),(2,-1,1,-5),(1,-2,-1,-4),(4,5,9,-3)\rangle.
\]

\begin{enumerate}[label=\alph*)]

\item Describe $V$ by linear equations; that is, find a linear system whose solution set is $V$. Use your answer to test whether $(5,-4,1,-6)\in V$ and whether $(3,2,7,-4)\in V$.

\begin{answer}
Let $v_1=(2,2,4,-2)$, $v_2=(2,-1,1,-5)$,
$v_3=(1,-2,-1,-4)$ and $v_4=(4,5,9,-3)$.

The relations
\[
 v_3=-\tfrac12v_1+v_2,\qquad v_4=\tfrac73v_1-\tfrac13v_2
\]
show that $V=\langle v_1,v_2\rangle$. Both $v_1$ and $v_2$ satisfy
\[
 x+y-z=0,\qquad 2x-y+w=0.
\]
Conversely, any vector satisfying these equations has $z=x+y$, $w=-2x+y$,
and
\[
 (x,y,x+y,-2x+y)
 =\frac{x+2y}{6}v_1+\frac{x-y}{3}v_2.
\]
Therefore
\[
 V=\{(x,y,z,w):x+y-z=0,\ 2x-y+w=0\}.
\]
For $(5,-4,1,-6)$ the second left side equals $8$, so it is not in $V$.
For $(3,2,7,-4)$ the first left side equals $-2$, so it is not in $V$ either.
\end{answer}

\item Let
\[
W=\{(x,y,z,w)\in\RR^4:3y-2z-w=0\ \land\ x+7y-5z-w=0\}.
\]
Using your answer to (a), determine a finite generating system for $V\cap W$.

\begin{answer}
Combine these two equations with those defining $W$:
\[
 x+y-z=0,\quad2x-y+w=0,\quad3y-2z-w=0,\quad x+7y-5z-w=0.
\]
Substitute $z=x+y$, $w=-2x+y$. The third equation becomes an identity;
the fourth becomes $-2x+y=0$. Thus $y=2x$, $z=3x$, $w=0$, and
\[
 V\cap W=\{(t,2t,3t,0):t\in\RR\}=\langle(1,2,3,0)\rangle.
\]
A finite generating system is $\bigl((1,2,3,0)\bigr)$.
The answer key gives the equivalent scaled system
\[
 \bigl((1/3,2/3,1,0)\bigr).
\]
\end{answer}
\end{enumerate}
