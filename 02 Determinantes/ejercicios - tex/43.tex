\item {\bfseries\boldmath
What is the determinant of each of the following matrices?
}

\begin{enumerate}[label=\alph*)]

\item $A=\begin{bmatrix}1&0&0&0\\1&-2&0&0\\2&3&0&0\\1&5&6&7\end{bmatrix}$

\begin{answer}
$A$ is lower triangular, so $\det A=1(-2)\cdot0\cdot7=0$.
\end{answer}

\item $B=\begin{bmatrix}1&0&0&0\\0&4&0&0\\0&0&9&0\\0&0&0&-6\end{bmatrix}$

\begin{answer}
$B$ is diagonal, so $\det B=1\cdot4\cdot9\cdot(-6)=-216$.
\end{answer}

\item $C=\begin{bmatrix}-6&9&-7\\0&2&11\\0&0&4\end{bmatrix}$

\begin{answer}
$C$ is upper triangular, so $\det C=(-6)\cdot2\cdot4=-48$.
\end{answer}

\item $D=\begin{bmatrix}3&-4&12&\sqrt2&6\\12&-2&6&-7&4\\0&0&0&0&0\\1&\sqrt3&5&5&9\\1&-3&5&-6&7\end{bmatrix}$

\begin{answer}
$D$ has a zero third row, so $\det D=0$.
\end{answer}

\item $E=\begin{bmatrix}5&0&-7\\0&0&12\\-8&0&9\end{bmatrix}$

\begin{answer}
$E$ has a zero second column, so $\det E=0$.
\end{answer}
\end{enumerate}
