\setcounter{enumi}{41}
\item {\bfseries\boldmath
Consider the matrix
}

\[
A=\begin{bmatrix}0&2&1\\2&-1&3\\0&1&4\end{bmatrix}.
\]
Compute $\det A$

\begin{enumerate}[label=\alph*)]

\item using Sarrus\textquotesingle{} rule.

\begin{answer}
Let $A=\begin{bmatrix}0&2&1\\2&-1&3\\0&1&4\end{bmatrix}$.

By Sarrus' rule,
\[
 \det A=(0\cdot(-1)\cdot4+2\cdot3\cdot0+1\cdot2\cdot1)
 -(1\cdot(-1)\cdot0+2\cdot2\cdot4+0\cdot3\cdot1)=-14.
\]
\end{answer}

\item by Laplace expansion along the second row.

\begin{answer}
Laplace expansion along row $2$ gives
\[
 \det A=-2\det\begin{bmatrix}2&1\\1&4\end{bmatrix}
 -\det\begin{bmatrix}0&1\\0&4\end{bmatrix}
 -3\det\begin{bmatrix}0&2\\0&1\end{bmatrix}=-2(8-1)=-14.
\]
\end{answer}

\item by Laplace expansion along the first column.

\begin{answer}
Expansion along column $1$ gives immediately
\[
 \det A=-2\det\begin{bmatrix}2&1\\1&4\end{bmatrix}=-14.
\]
\end{answer}
\end{enumerate}
