% ---------- Exercise 55 ----------
\setcounter{enumi}{54}
\item {\bfseries\boldmath
Let
}

\[
A=\begin{bmatrix}2&1&0\\1&0&1\\2&2&0\end{bmatrix}.
\]

\begin{enumerate}[label=\alph*)]

\item Compute $\det A$ and conclude that $A$ is invertible.

\begin{answer}
Let $A=\begin{bmatrix}2&1&0\\1&0&1\\2&2&0\end{bmatrix}$.

Expansion along row $1$ gives
$\det A=2(0-2)-(0-2)=-2\ne0$, so $A$ is invertible.
\end{answer}

\item Determine $\widehat A$, the matrix of cofactors of $A$.

\begin{answer}
If $A_{ij}$ is obtained by deleting row $i$ and column $j$, the cofactor
matrix is $\widehat A=[(-1)^{i+j}\det A_{ij}]$. Computing its entries gives
\[
 \widehat A=
 \begin{bmatrix}
 0-2&-(0-2)&2-0\\-(0-0)&0-0&-(4-2)\\1-0&-(2-0)&0-1
 \end{bmatrix}
 =\begin{bmatrix}-2&2&2\\0&0&-2\\1&-2&-1\end{bmatrix}.
\]
\end{answer}

\item Compute $A^{-1}$ using part (b).

\begin{answer}
The adjugate is the transpose of the cofactor matrix. Hence
\[
 A^{-1}=\frac{\widehat A^{\,T}}{\det A}
 =-\tfrac12\begin{bmatrix}-2&0&1\\2&0&-2\\2&-2&-1\end{bmatrix}
 =\begin{bmatrix}1&0&-1/2\\-1&0&1\\-1&1&1/2\end{bmatrix}.
\]
\end{answer}
\end{enumerate}
