\setcounter{enumi}{47}
\item {\bfseries\boldmath
Consider the following real matrix:
}

\[
A=\begin{bmatrix}a&b&c\\d&e&f\\g&h&i\end{bmatrix}.
\]
Knowing that $\det A=2$, compute the determinants of the following matrices.

\begin{enumerate}[label=\alph*)]

\item $B=\begin{bmatrix}a&2b&c\\3d&6e&3f\\-2g&-4h&-2i\end{bmatrix}$.

\begin{answer}
Let $A=\begin{bmatrix}a&b&c\\d&e&f\\g&h&i\end{bmatrix}$, with $\det A=2$.

$B$ is obtained by multiplying row $2$ by $3$, row $3$ by $-2$,
and column $2$ by $2$. Therefore $\det B=3(-2)2\det A=-24$.
\end{answer}

\item $C=4A$.

\begin{answer}
Since $A$ has order $3$, $\det(4A)=4^3\det A=128$.
\end{answer}

\item $D=\begin{bmatrix}a&b&b&c\\2&0&3&0\\d&e&e&f\\g&h&h&i\end{bmatrix}$.

\begin{answer}
In
$D=\begin{bmatrix}a&b&b&c\\2&0&3&0\\d&e&e&f\\g&h&h&i\end{bmatrix}$,
replace column $3$ by column $3$ minus column $2$:
\[
 \det D=\det\begin{bmatrix}a&b&0&c\\2&0&3&0\\d&e&0&f\\g&h&0&i\end{bmatrix}
 =3(-1)^{2+3}\det A=-6.
\]
\end{answer}
\end{enumerate}
