\setcounter{enumi}{55}
\item {\bfseries\boldmath
Let
}

\[
A=\begin{bmatrix}4&3&1&6\\0&-3&-3&1\\4&1&0&14\\4&-1&-2&2\end{bmatrix}.
\]

\begin{enumerate}[label=\alph*)]

\item Compute $\det A$ and conclude that $A$ is invertible.

\begin{answer}
\[
 A=\begin{bmatrix}4&3&1&6\\0&-3&-3&1\\4&1&0&14\\4&-1&-2&2\end{bmatrix}.
\]

Use $F_3\gets F_3-F_1$, $F_4\gets F_4-F_1$ and expand along column $1$:
\[
 \det A=4\det\begin{bmatrix}-3&-3&1\\-2&-1&8\\-4&-3&-4\end{bmatrix}
 =4[-3(28)+3(40)+2]=152\ne0.
\]
Thus $A$ is invertible.
\end{answer}

\item Without computing the whole matrix $A^{-1}$, compute its entries in positions $(2,3)$ and $(3,2)$.

\begin{answer}
The entry $(i,j)$ of $A^{-1}$ is the cofactor in position $(j,i)$ of
$A$, divided by $\det A$. Therefore
\[
\begin{aligned}
 (A^{-1})_{23}
 &=\frac{-\det\begin{bmatrix}4&1&6\\0&-3&1\\4&-2&2\end{bmatrix}}{152}
 =\frac{-60}{152}=-\frac{15}{38},\\[4pt]
 (A^{-1})_{32}
 &=\frac{-\det\begin{bmatrix}4&3&6\\4&1&14\\4&-1&2\end{bmatrix}}{152}
 =\frac{-160}{152}=-\frac{20}{19}.
\end{aligned}
\]
Only the two required minors have been computed.
\end{answer}
\end{enumerate}
