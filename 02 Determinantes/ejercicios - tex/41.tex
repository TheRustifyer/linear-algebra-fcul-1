\item {\bfseries\boldmath
Compute the determinants of the following matrices.
}

\begin{enumerate}[label=\alph*)]

\item $\begin{bmatrix}3&1\\2&4\end{bmatrix}$

\begin{answer}
$\det\begin{bmatrix}3&1\\2&4\end{bmatrix}=3\cdot4-1\cdot2=10$.
\end{answer}

\item $\begin{bmatrix}6&-12\\-2&4\end{bmatrix}$

\begin{answer}
$\det\begin{bmatrix}6&-12\\-2&4\end{bmatrix}=24-24=0$.
\end{answer}

\item $\begin{bmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{bmatrix}$

\begin{answer}
$\det\begin{bmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{bmatrix}
=\cos^2\theta+\sin^2\theta=1$.
\end{answer}

\item $\begin{bmatrix}0&2&2\\1&0&3\\2&1&1\end{bmatrix}$

\begin{answer}
Expanding along the first row,
\[
 \det\begin{bmatrix}0&2&2\\1&0&3\\2&1&1\end{bmatrix}
 =-2(1-6)+2(1-0)=12.
\]
\end{answer}
\end{enumerate}
