\setcounter{enumi}{51}
\item {\bfseries\boldmath
Show that for all $A,B\in M_n(\KK)$, $\det(AB)=\det(BA)$.
}

\begin{answer}
For $A,B\in M_n(\KK)$,
\[
 \det(AB)=\det A\det B=\det B\det A=\det(BA),
\]
since determinants are scalars and scalar multiplication is commutative.
\end{answer}
