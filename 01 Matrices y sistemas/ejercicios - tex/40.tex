\setcounter{enumi}{39}
\item {\bfseries\boldmath
Let $A\in M_n(\CC)$ satisfy $AA^*=3I_n$.
}

\begin{enumerate}[label=\alph*)]

\item Show that $A$ is invertible and determine its inverse in terms of $A$.

\begin{answer}
Assume $A\in M_n(\CC)$ and $AA^*=3I_n$.

$A(\frac13A^*)=I_n$. A square matrix with a right inverse is invertible;
hence $A^{-1}=\frac13A^*$.
For completeness, the right inverse gives a solution of $AX=Y$ for every
column $Y$, so the echelon form of $A$ has a pivot in every row. Since $A$
is square it reduces to $I_n$, proving invertibility.
\end{answer}

\item Justify that $\overline A$ is invertible and determine its inverse in terms of $A$.

\begin{answer}
By Exercise 23, $\overline A$ is invertible and
\[
 (\overline A)^{-1}=\overline{A^{-1}}
 =\tfrac13\overline{A^*}=\tfrac13 A^T.
\]
\end{answer}
\end{enumerate}
