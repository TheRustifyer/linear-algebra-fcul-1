% ---------- Exercise 26 ----------
\setcounter{enumi}{25}
\item {\bfseries\boldmath
For each of the following matrices, determine whether it is invertible and, if it is, compute its inverse.
}

\begin{enumerate}[label=\alph*)]

\item $A=\begin{bmatrix}1&2\\4&8\end{bmatrix}$

\begin{answer}
For the row reductions below, apply the stated operations to $[M\mid I]$.
When the left block becomes $I$, the right block is $M^{-1}$.

$A=\begin{bmatrix}1&2\\4&8\end{bmatrix}$ is not invertible:
$F_2\gets F_2-4F_1$ produces a zero row, so $\rank(A)=1<2$.
\end{answer}

\item $B=\begin{bmatrix}1&1\\1&-2\end{bmatrix}$

\begin{answer}
For $B=\begin{bmatrix}1&1\\1&-2\end{bmatrix}$, use
$F_2\gets F_2-F_1$, $F_2\gets-\frac13F_2$, $F_1\gets F_1-F_2$.
Then
\[
 B^{-1}=\begin{bmatrix}2/3&1/3\\1/3&-1/3\end{bmatrix}.
\]
\end{answer}

\item $C=\begin{bmatrix}5&0&0\\0&-4&0\\0&0&7\end{bmatrix}$

\begin{answer}
$C=\operatorname{diag}(5,-4,7)$ is invertible. Dividing its three rows
by $5,-4,7$ gives
$C^{-1}=\operatorname{diag}(\frac15,-\frac14,\frac17)$.
\end{answer}

\item $D=\begin{bmatrix}0&0&5\\0&-4&0\\7&0&0\end{bmatrix}$

\begin{answer}
For $D=\begin{bmatrix}0&0&5\\0&-4&0\\7&0&0\end{bmatrix}$,
interchange rows $1,3$ and divide the resulting rows by $7,-4,5$.
Thus
\[
 D^{-1}=\begin{bmatrix}0&0&1/7\\0&-1/4&0\\1/5&0&0\end{bmatrix}.
\]
\end{answer}

\item $E=\begin{bmatrix}1&0&4\\0&1&0\\0&0&1\end{bmatrix}$

\begin{answer}
$E=\begin{bmatrix}1&0&4\\0&1&0\\0&0&1\end{bmatrix}$ is invertible.
The operation $F_1\gets F_1-4F_3$ gives
$E^{-1}=\begin{bmatrix}1&0&-4\\0&1&0\\0&0&1\end{bmatrix}$.
\end{answer}

\item $F=\begin{bmatrix}0&1&0\\1&0&0\\0&0&1\end{bmatrix}$

\begin{answer}
$F=\begin{bmatrix}0&1&0\\1&0&0\\0&0&1\end{bmatrix}$ satisfies
$F^2=I_3$, so $F^{-1}=F$.
\end{answer}

\item $G=\begin{bmatrix}0&3&1\\-1&3&1\\0&1&1\end{bmatrix}$

\begin{answer}
For $G=\begin{bmatrix}0&3&1\\-1&3&1\\0&1&1\end{bmatrix}$, use
\[
\begin{gathered}
 F_1\leftrightarrow F_2,\quad F_1\gets-F_1,\quad
 F_2\leftrightarrow F_3,\quad F_3\gets F_3-3F_2,\quad F_3\gets-\tfrac12F_3,\\
 F_2\gets F_2-F_3,\quad F_1\gets F_1+F_3,\quad F_1\gets F_1+3F_2.
\end{gathered}
\]
The left block becomes $I_3$, and
\[
 G^{-1}=\begin{bmatrix}1&-1&0\\1/2&0&-1/2\\-1/2&0&3/2\end{bmatrix}.
\]
\end{answer}

\item $H=\begin{bmatrix}2&-1&4\\1&-1&2\\-1&0&-2\end{bmatrix}$

\begin{answer}
$H=\begin{bmatrix}2&-1&4\\1&-1&2\\-1&0&-2\end{bmatrix}$ is not invertible.
Its third column is twice its first column, so
$H\begin{bmatrix}-2&0&1\end{bmatrix}^T=0$ with a nonzero column.
If $H^{-1}$ existed, multiplication by it would give a contradiction.
\end{answer}

\item $I=\begin{bmatrix}2&-1&4\\1&-1&2\\-1&0&-3\end{bmatrix}$

\begin{answer}
For $I=\begin{bmatrix}2&-1&4\\1&-1&2\\-1&0&-3\end{bmatrix}$, use
\[
\begin{gathered}
 F_1\gets\tfrac12F_1,\quad F_2\gets F_2-F_1,\quad
 F_3\gets F_3+F_1,\quad F_2\gets-2F_2,\\
 F_3\gets F_3+\tfrac12F_2,\quad F_3\gets-F_3,\quad
 F_1\gets F_1-2F_3,\quad F_1\gets F_1+\tfrac12F_2.
\end{gathered}
\]
Hence $I$ is invertible and
$I^{-1}=\begin{bmatrix}3&-3&2\\1&-2&0\\-1&1&-1\end{bmatrix}$.
\end{answer}

\item $J=\begin{bmatrix}3&1&0\\1&2&1\\2&-1&-1\end{bmatrix}$

\begin{answer}
$J=\begin{bmatrix}3&1&0\\1&2&1\\2&-1&-1\end{bmatrix}$ is not invertible:
its first row is the sum of rows $2,3$. Equivalently,
$J\begin{bmatrix}1&-3&5\end{bmatrix}^T=0$ with a nonzero column.
\end{answer}

\item $K=\begin{bmatrix}1&-1&1&2\\2&-2&1&1\\1&-1&0&1\\-2&0&2&-2\end{bmatrix}$

\begin{answer}
For $K=\begin{bmatrix}1&-1&1&2\\2&-2&1&1\\1&-1&0&1\\-2&0&2&-2\end{bmatrix}$,
apply
\[
\begin{gathered}
 F_2\gets F_2-2F_1,\quad F_3\gets F_3-F_1,\quad F_4\gets F_4+2F_1,\quad
 F_2\leftrightarrow F_4,\\
 F_2\gets-\tfrac12F_2,\quad F_3\gets-F_3,\quad
 F_4\gets F_4+F_3,\quad F_4\gets-\tfrac12F_4.
\end{gathered}
\]
The augmented matrix is now
\[
\left[\begin{array}{rrrr|rrrr}
1&-1&1&2&1&0&0&0\\
0&1&-2&-1&-1&0&0&-1/2\\
0&0&1&1&1&0&-1&0\\
0&0&0&1&1/2&-1/2&1/2&0
\end{array}\right].
\]
Complete the reduction with
\[
\begin{gathered}
 F_1\gets F_1-2F_4,\quad F_2\gets F_2+F_4,\quad F_3\gets F_3-F_4,\\
 F_1\gets F_1-F_3,\quad F_2\gets F_2+2F_3,\quad F_1\gets F_1+F_2.
\end{gathered}
\]
Thus
\[
 K^{-1}=\begin{bmatrix}
0&1&-2&-1/2\\1/2&1/2&-5/2&-1/2\\
1/2&1/2&-3/2&0\\1/2&-1/2&1/2&0
\end{bmatrix}.
\]
\end{answer}
\end{enumerate}
