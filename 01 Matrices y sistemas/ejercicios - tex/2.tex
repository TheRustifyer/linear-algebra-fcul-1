\item  Write the matrix $\left(A=\left[a_{ij}\right]_{3\times3}\right)$ defined by:
    \begin {enumerate}
        \item $\mathlarger{a}_{ij} = i + j$
        \begin{answer}
            $$
            A = \begin{pmatrix}
            2 & 3 & 4 \\
            3 & 4 & 5 \\
            4 & 5 & 6
            \end{pmatrix}
            $$
        \end{answer}
        \item $\mathlarger{a}_{ij} = \begin{cases}
                0, & \text{if } i = j,\\
                -1, & \text{if } i > j,\\
                2, & \text{if } i < j.
                \end{cases}$
        \begin{answer}
            $$
            A = \begin{pmatrix}
            0 & 2 & 2 \\
            -1 & 0 & 2 \\
            -1 & -1 & 0
            \end{pmatrix}
            $$
        \end{answer}
    \end{enumerate}
