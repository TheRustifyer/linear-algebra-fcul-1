\item \textbf{Calculate the inverse of the matrix}

$$
A=
\begin{bmatrix}
    1 & 0 & i\\
    0 & 2i & -1\\
    2 & -4 & i
\end{bmatrix}
$$

\begin{answer}
\end{answer}

\begin{answer}
    {\color{red}
    \textbf{Note:} The imaginary unit satisfies
    $i^2=-1$, so $\frac{1}{i}=-i$ and
    $\frac{1}{2i}=-\frac{i}{2}$
    }

    $$
    [A\mid I_3]=
    \left[
    \begin{array}{ccc|ccc}
        1 & 0 & i & 1 & 0 & 0\\
        0 & 2i & -1 & 0 & 1 & 0\\
        2 & -4 & i & 0 & 0 & 1
    \end{array}
    \right]
    \xrightarrow{R_3\leftarrow R_3-2R_1}
    \left[
    \begin{array}{ccc|ccc}
        1 & 0 & i & 1 & 0 & 0\\
        0 & 2i & -1 & 0 & 1 & 0\\
        0 & -4 & -i & -2 & 0 & 1
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_3\leftarrow R_3-2iR_2}
    \left[
    \begin{array}{ccc|ccc}
        1 & 0 & i & 1 & 0 & 0\\
        0 & 2i & -1 & 0 & 1 & 0\\
        0 & 0 & i & -2 & -2i & 1
    \end{array}
    \right]
    \xrightarrow{R_3\leftarrow-iR_3}
    \left[
    \begin{array}{ccc|ccc}
        1 & 0 & i & 1 & 0 & 0\\
        0 & 2i & -1 & 0 & 1 & 0\\
        0 & 0 & 1 & 2i & -2 & -i
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_1\leftarrow R_1-iR_3}
    \left[
    \begin{array}{ccc|ccc}
        1 & 0 & 0 & 3 & 2i & -1\\
        0 & 2i & -1 & 0 & 1 & 0\\
        0 & 0 & 1 & 2i & -2 & -i
    \end{array}
    \right]
    \xrightarrow{R_2\leftarrow R_2+R_3}
    \left[
    \begin{array}{ccc|ccc}
        1 & 0 & 0 & 3 & 2i & -1\\
        0 & 2i & 0 & 2i & -1 & -i\\
        0 & 0 & 1 & 2i & -2 & -i
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_2\leftarrow-\frac{i}{2}R_2}
    \left[
    \begin{array}{ccc|ccc}
        1 & 0 & 0 & 3 & 2i & -1\\
        0 & 1 & 0 & 1 & \frac{i}{2} & -\frac12\\
        0 & 0 & 1 & 2i & -2 & -i
    \end{array}
    \right]
    \implies
    $$

    $$
    \boxed{
    A^{-1}=
    \begin{bmatrix}
        3 & 2i & -1\\
        1 & \frac{i}{2} & -\frac12\\
        2i & -2 & -i
    \end{bmatrix}
    }
    $$
\end{answer}
