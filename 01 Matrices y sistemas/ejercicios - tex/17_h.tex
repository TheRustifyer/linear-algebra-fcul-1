$H=
\begin{bmatrix}
    3 & 0 & -1 & 0 & 1
\end{bmatrix}
$ \\[1.0em]

\begin{enumerate}
    \item
    \textbf{Row Echelon:}
    \begin{answer}
        $H$ is already in row echelon form.
    \end{answer}

    \item
    \textbf{Rank:}
    \begin{answer}
        $\operatorname{rank}(H)=1.$
    \end{answer} \\

    \item
    \textbf{Reduced Echelon form:} \\[1.0em]
    \begin{answer}
        $H \xrightarrow[t_1]{R_1 \leftarrow \frac{1}{3}R_1}
        \begin{bmatrix}
            1 & 0 & -\frac{1}{3} & 0 & \frac{1}{3}
        \end{bmatrix}
        = \operatorname{rref}(H)$
    \end{answer}
\end{enumerate}
