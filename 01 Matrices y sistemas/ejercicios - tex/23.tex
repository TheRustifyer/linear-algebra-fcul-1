\setcounter{enumi}{22}
\item {\bfseries\boldmath
Show that if $A\in M_n(\KK)$ is invertible, then $\overline A$ is also invertible and $(\overline A)^{-1}=\overline{A^{-1}}$.
}

\begin{answer}
Conjugating $AA^{-1}=A^{-1}A=I_n$ gives
\[
 \overline A\,\overline{A^{-1}}=I_n
 =\overline{A^{-1}}\,\overline A.
\]
Hence $\overline A$ is invertible and
$(\overline A)^{-1}=\overline{A^{-1}}$.
\end{answer}
