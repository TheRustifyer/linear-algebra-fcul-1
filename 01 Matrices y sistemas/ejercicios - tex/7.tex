\item Let the following matrices be defined:

$$
A = \begin{bmatrix}
    3 \\
    2 \\
    -1 \\
    1
\end{bmatrix},
\qquad
B = \begin{bmatrix}
    -1 & 0 & 1 & 2
\end{bmatrix}.
$$

Compute $AB$ and $BA$.\\[0.5em]

\textbf{Solution:}

\begin{answer}
    Given $A \in \mathcal{M}_{4\times1}(\mathbb{R})$
    and $B \in \mathcal{M}_{1\times4}(\mathbb{R})$,
    both products are defined.

    $$
    AB =
    \begin{bmatrix}
        3(-1) & 3(0) & 3(1) & 3(2) \\
        2(-1) & 2(0) & 2(1) & 2(2) \\
        (-1)(-1) & (-1)(0) & (-1)(1) & (-1)(2) \\
        1(-1) & 1(0) & 1(1) & 1(2)
    \end{bmatrix}
    =
    \begin{bmatrix}
        -3 & 0 & 3 & 6 \\
        -2 & 0 & 2 & 4 \\
        1 & 0 & -1 & -2 \\
        -1 & 0 & 1 & 2
    \end{bmatrix}.
    $$

    $$
    \begin{aligned}
    BA
        &= \begin{bmatrix}
            \displaystyle\sum_{k=1}^{4}b_{1k}a_{k1}
        \end{bmatrix} \\[1em]
        &= \begin{bmatrix}
            (-1)(3)+0(2)+1(-1)+2(1)
        \end{bmatrix} \\[1em]
        &= \boxed{\begin{bmatrix}-2\end{bmatrix}}.
    \end{aligned}
    $$
\end{answer}
