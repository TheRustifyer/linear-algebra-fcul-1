$F=
\begin{bmatrix}
    3 & 0 & 0 & -21 & 24 \\
    4 & 1 & 0 & -25 & 34 \\
   -2 & 3 & 0 & 22 & -2
\end{bmatrix}
$ \\[1.0em]

\begin{enumerate}
    \item
    \textbf{Row Echelon:} \\[1.0em]
    \begin{answer}
        $F \xrightarrow[t_1]{
            \substack{
                R_2 \leftarrow R_2-R_1 \\
                R_1 \leftrightarrow R_2
            }
        }
        \begin{bmatrix}
            1 & 1 & 0 & -4 & 10 \\
            3 & 0 & 0 & -21 & 24 \\
           -2 & 3 & 0 & 22 & -2
        \end{bmatrix}
        $ \\[1.0em]

        $\xrightarrow[t_2]{
            \substack{
                R_2 \leftarrow R_2-3R_1 \\
                R_3 \leftarrow R_3+2R_1
            }
        }
        \begin{bmatrix}
            1 & 1 & 0 & -4 & 10 \\
            0 & -3 & 0 & -9 & -6 \\
            0 & 5 & 0 & 14 & 18
        \end{bmatrix}
        $
        $\xrightarrow[t_3]{
            \substack{
                R_2 \leftarrow -\frac{1}{3}R_2 \\
                R_3 \leftarrow R_3-5R_2
            }
        }
        \begin{bmatrix}
            1 & 1 & 0 & -4 & 10 \\
            0 & 1 & 0 & 3 & 2 \\
            0 & 0 & 0 & -1 & 8
        \end{bmatrix}
        = F'$
    \end{answer}

    \item
    \textbf{Rank:}
    \begin{answer}
        $\operatorname{rank}(F)=3.$
    \end{answer} \\

    \item
    \textbf{Reduced Echelon form:} \\[1.0em]
    \begin{answer}
        $F' \xrightarrow[t_4]{R_3 \leftarrow -R_3}
        \begin{bmatrix}
            1 & 1 & 0 & -4 & 10 \\
            0 & 1 & 0 & 3 & 2 \\
            0 & 0 & 0 & 1 & -8
        \end{bmatrix}
        $ \\[1.0em]

        $\xrightarrow[t_5]{
            \substack{
                R_2 \leftarrow R_2-3R_3 \\
                R_1 \leftarrow R_1+4R_3 \\
                R_1 \leftarrow R_1-R_2
            }
        }
        \begin{bmatrix}
            1 & 0 & 0 & 0 & -48 \\
            0 & 1 & 0 & 0 & 26 \\
            0 & 0 & 0 & 1 & -8
        \end{bmatrix}
        = \operatorname{rref}(F)$
    \end{answer}
\end{enumerate}
