\setcounter{enumi}{19}
\item {\bfseries\boldmath
Consider the matrices
}

\[
A=\begin{bmatrix}3&5\\1&2\end{bmatrix},\quad B=\begin{bmatrix}2&-5\\-1&3\end{bmatrix},\quad C=\begin{bmatrix}0&-1\\1&-1\end{bmatrix}.
\]

\begin{enumerate}[label=\alph*)]

\item Verify that $B$ is the inverse of $A$.

\begin{answer}
Here $A=\begin{bmatrix}3&5\\1&2\end{bmatrix}$,
$B=\begin{bmatrix}2&-5\\-1&3\end{bmatrix}$,
$C=\begin{bmatrix}0&-1\\1&-1\end{bmatrix}$.

\[
 AB=\begin{bmatrix}6-5&-15+15\\2-2&-5+6\end{bmatrix}=I_2,
 \quad BA=\begin{bmatrix}6-5&10-10\\-3+3&-5+6\end{bmatrix}=I_2.
\]
Thus $B=A^{-1}$.
\end{answer}

\item Is the matrix $\begin{bmatrix}2&-5\\-1&2\end{bmatrix}$ the inverse of $A$? Justify your answer.

\begin{answer}
For $D=\begin{bmatrix}2&-5\\-1&2\end{bmatrix}$,
$AD=\begin{bmatrix}1&-5\\0&-1\end{bmatrix}\ne I_2$, so $D\ne A^{-1}$.
\end{answer}

\item Determine $(5A)^{-1}$ and $(A^T)^{-1}$.

\begin{answer}
\[
 (5A)^{-1}=\tfrac15B=\begin{bmatrix}2/5&-1\\-1/5&3/5\end{bmatrix},\qquad
 (A^T)^{-1}=B^T=\begin{bmatrix}2&-1\\-5&3\end{bmatrix}.
\]
\end{answer}

\item Compute $C^3$. Is $C$ invertible? What is its inverse?

\begin{answer}
\[
 C^2=\begin{bmatrix}-1&1\\-1&0\end{bmatrix},\quad C^3=I_2.
\]
Therefore $C^{-1}=C^2$, since $CC^2=C^2C=I_2$.
\end{answer}

\item Is $AC$ invertible? Justify your answer and, if it is, determine $(AC)^{-1}$.

\begin{answer}
The product is invertible, with
\[
 (AC)^{-1}=C^{-1}A^{-1}=C^2B=\begin{bmatrix}-3&8\\-2&5\end{bmatrix}.
\]
Indeed $(AC)(C^{-1}A^{-1})=I_2=(C^{-1}A^{-1})(AC)$.
\end{answer}
\end{enumerate}
