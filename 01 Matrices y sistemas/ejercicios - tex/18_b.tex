\textbf{Determine, if they exist, elementary matrices
$E_p,\ldots,E_1$ such that}

\begin{enumerate}[label=\roman*)]
    \item
    $$
    E_p\cdots E_1A=
    \begin{bmatrix}
        1 & 0 & 1 & 1 \\
        0 & -1 & 1 & 2 \\
        0 & 2 & 2 & 0
    \end{bmatrix}.
    $$

    \begin{answer}
        $A \xrightarrow[t_1]{R_2 \leftarrow R_2-R_1}
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & 2 & 2 & 0 \\
        -2 & -1 & -1 & 0
        \end{bmatrix}
        $
        $\xrightarrow[t_2]{R_3 \leftarrow R_3+2R_1}
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & 2 & 2 & 0 \\
            0 & -1 & 1 & 2
        \end{bmatrix}
        $ \\[1.0em]

        $\xrightarrow[t_3]{R_2 \leftrightarrow R_3}
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & -1 & 1 & 2 \\
            0 & 2 & 2 & 0
        \end{bmatrix}.
        $ \\[1.0em]

        The corresponding elementary matrices are

        $$
        \begin{aligned}
        E_1 &=
        \begin{bmatrix}
            1 & 0 & 0 \\
        -1 & 1 & 0 \\
            0 & 0 & 1
        \end{bmatrix},
        &
        E_2 &=
        \begin{bmatrix}
            1 & 0 & 0 \\
            0 & 1 & 0 \\
            2 & 0 & 1
        \end{bmatrix},
        &
        E_3 &=
        \begin{bmatrix}
            1 & 0 & 0 \\
            0 & 0 & 1 \\
            0 & 1 & 0
        \end{bmatrix}.
        \end{aligned}
        $$

        Hence,

        $$
        \boxed{
        E_3E_2E_1A=
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & -1 & 1 & 2 \\
            0 & 2 & 2 & 0
        \end{bmatrix}
        }.
        $$
    \end{answer}

    \item
    $$
    E_p\cdots E_1A=
    \begin{bmatrix}
        1 & 0 & 1 & 1 \\
        0 & 3 & -3 & -6 \\
        0 & 0 & 7 & 7
    \end{bmatrix}.
    $$

    \begin{answer}
        We reuse the first three elementary
        transformations from i):

        $E_3E_2E_1A=
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & -1 & 1 & 2 \\
            0 & 2 & 2 & 0
        \end{bmatrix}
        $ \\[1.0em]

        $\xrightarrow[t_4]{R_2 \leftarrow -3R_2}
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & 3 & -3 & -6 \\
            0 & 2 & 2 & 0
        \end{bmatrix}
        $
        $\xrightarrow[t_5]{R_3 \leftarrow R_3-\frac{2}{3}R_2}
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & 3 & -3 & -6 \\
            0 & 0 & 4 & 4
        \end{bmatrix}
        $ \\[1.0em]

        $\xrightarrow[t_6]{R_3 \leftarrow \frac{7}{4}R_3}
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & 3 & -3 & -6 \\
            0 & 0 & 7 & 7
        \end{bmatrix}.
        $ \\[1.0em]

        The additional elementary matrices are

        $$
        \begin{aligned}
        E_4 &=
        \begin{bmatrix}
            1 & 0 & 0 \\
            0 & -3 & 0 \\
            0 & 0 & 1
        \end{bmatrix},
        &
        E_5 &=
        \begin{bmatrix}
            1 & 0 & 0 \\
            0 & 1 & 0 \\
            0 & -\frac{2}{3} & 1
        \end{bmatrix},
        &
        E_6 &=
        \begin{bmatrix}
            1 & 0 & 0 \\
            0 & 1 & 0 \\
            0 & 0 & \frac{7}{4}
        \end{bmatrix}.
        \end{aligned}
        $$

        Therefore,

        $$
        \boxed{
        E_6E_5E_4E_3E_2E_1A=
        \begin{bmatrix}
            1 & 0 & 1 & 1 \\
            0 & 3 & -3 & -6 \\
            0 & 0 & 7 & 7
        \end{bmatrix}
        }.
        $$
    \end{answer}

\end{enumerate}
