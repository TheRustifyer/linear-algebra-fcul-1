% ---------- Exercise 18 ----------
\setcounter{enumi}{17}
\item {\bfseries\boldmath
Consider the following matrices.
}

\[
A=\begin{bmatrix}1&0&1&1\\1&2&3&1\\-2&-1&-1&0\end{bmatrix},\quad B=\begin{bmatrix}1&2&3&1\\1&0&1&1\\-2&-1&-1&0\end{bmatrix}
\]

\[
C=\begin{bmatrix}1&0&1&1\\-1&1&2&1\\-2&-1&-1&0\end{bmatrix},\quad D=\begin{bmatrix}1&0&1&1\\-2&-4&-6&-2\\-2&-1&-1&0\end{bmatrix}
\]

\begin{enumerate}[label=\alph*)]

\item Determine elementary matrices $H,J,K$ such that $HA=B$, $JA=C$ and $KA=D$.

\begin{answer}
Let $T_{jk}(c)$ be $I_3$ with entry $(j,k)$ replaced by $c$, for $j\ne k$.
Left multiplication by $T_{jk}(c)$ performs $F_j\gets F_j+cF_k$.
Let $S_{23}$ interchange rows $2,3$ and let $D_j(c)$ multiply row $j$ by $c$.
These are elementary matrices when $c\ne0$ in $D_j(c)$.
Here
\[
 A=\begin{bmatrix}1&0&1&1\\1&2&3&1\\-2&-1&-1&0\end{bmatrix}.
\]

$B$ is obtained by interchanging rows $1,2$; $C$ by
$F_2\gets F_2+F_3$; $D$ by $F_2\gets-2F_2$. Therefore
\[
 H=\begin{bmatrix}0&1&0\\1&0&0\\0&0&1\end{bmatrix},\quad
 J=\begin{bmatrix}1&0&0\\0&1&1\\0&0&1\end{bmatrix},\quad
 K=\begin{bmatrix}1&0&0\\0&-2&0\\0&0&1\end{bmatrix}.
\]
Direct multiplication gives $HA=B$, $JA=C$, $KA=D$.
\end{answer}

\item Determine, if they exist, elementary matrices $E_p,\ldots,E_1$ such that

\begin{enumerate}[label=\roman*)]

\item
\[
E_p\cdots E_1A=\begin{bmatrix}1&0&1&1\\0&-1&1&2\\0&2&2&0\end{bmatrix}.
\]

\begin{answer}
Apply $F_2\gets F_2-F_1$, $F_3\gets F_3+2F_1$ and
$F_2\leftrightarrow F_3$. Thus
\[
 S_{23}T_{31}(2)T_{21}(-1)A
 =\begin{bmatrix}1&0&1&1\\0&-1&1&2\\0&2&2&0\end{bmatrix}.
\]
Take $E_1=T_{21}(-1)$, $E_2=T_{31}(2)$, $E_3=S_{23}$.
\end{answer}

\item
\[
E_p\cdots E_1A=\begin{bmatrix}1&0&1&1\\0&3&-3&-6\\0&0&7&7\end{bmatrix}.
\]

\begin{answer}
Continue from (i) with $F_3\gets F_3+2F_2$,
$F_2\gets-3F_2$, $F_3\gets\frac74F_3$. Hence take
\[
\begin{gathered}
 E_1=T_{21}(-1),\quad E_2=T_{31}(2),\quad E_3=S_{23},\\
 E_4=T_{32}(2),\quad E_5=D_2(-3),\quad E_6=D_3(7/4).
\end{gathered}
\]
Indeed,
\[
 E_6\cdots E_1A
 =\begin{bmatrix}1&0&1&1\\0&3&-3&-6\\0&0&7&7\end{bmatrix}.
\]
The handwritten resolution prints $0$ in position $(1,4)$ of its final matrix.
The target in the exercise sheet and the stated operations give $1$ there.
\end{answer}
\end{enumerate}

\item Determine a matrix $Q$ such that $QA=F$, where
\[
F=\begin{bmatrix}-2&4&2&-2\\0&2&2&0\\0&3&9&6\end{bmatrix}.
\]

\begin{answer}
Apply to $[A\mid I_3]$ the operations
\[
\begin{gathered}
 F_2\gets F_2-F_1,\quad F_3\gets F_3+2F_1,\quad
 F_1\gets F_1-F_2,\\
 F_3\gets F_3+F_2,\quad F_1\gets-2F_1,\quad F_3\gets3F_3.
\end{gathered}
\]
The result is
\[
\left[\begin{array}{rrrr|rrr}
-2&4&2&-2&-4&2&0\\0&2&2&0&-1&1&0\\0&3&9&6&3&3&3
\end{array}\right].
\]
Consequently
$Q=\begin{bmatrix}-4&2&0\\-1&1&0\\3&3&3\end{bmatrix}$ satisfies $QA=F$.
\end{answer}
\end{enumerate}
