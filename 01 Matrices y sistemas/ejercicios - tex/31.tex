\setcounter{enumi}{30}
\item {\bfseries\boldmath
Determine whether each of the following equations is linear in the unknowns $x,y,z$.
}

\begin{enumerate}[label=\alph*)]

\item $x+\pi y+\sqrt[3]{2}\,z=e$.

\begin{answer}
An equation is linear in $x,y,z$ if it has the form $ax+by+cz=d$ with fixed
coefficients and no products or nonlinear functions of the unknowns.

$x+\pi y+\sqrt[3]{2}\,z=e$: linear; all coefficients are constants.
\end{answer}

\item $x^2+xy+z=2$.

\begin{answer}
$x^2+xy+z=2$: not linear, because of $x^2$ and $xy$.
\end{answer}

\item $x+\sin y+2z=0$.

\begin{answer}
$x+\sin y+2z=0$: not linear, because of $\sin y$.
\end{answer}

\item $4x+(\sin\frac\pi{10})^2y+23z=\cos\frac\pi5$.

\begin{answer}
$4x+(\sin\frac\pi{10})^2y+23z=\cos\frac\pi5$: linear; the
trigonometric expressions and $23$ are constants.
\end{answer}

\item $(-x+8y-2z)^2=1$.

\begin{answer}
$(-x+8y-2z)^2=1$: not linear. It is equivalent to the union of the
two equations $-x+8y-2z=1$ and $-x+8y-2z=-1$, not to a single linear equation.
\end{answer}

\item $ax+6y-3z=9$, with $a\in\RR$ fixed.

\begin{answer}
$ax+6y-3z=9$, with fixed $a\in\RR$: linear, including when $a=0$.
\end{answer}
\end{enumerate}
