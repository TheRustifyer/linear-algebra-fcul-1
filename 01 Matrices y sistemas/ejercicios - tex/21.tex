\setcounter{enumi}{20}
\item {\bfseries\boldmath
Consider two diagonal matrices of order $n$:
}

\[
A=\begin{bmatrix}a_1&0&\cdots&0\\0&a_2&\cdots&0\\\vdots&\vdots&\ddots&\vdots\\0&0&\cdots&a_n\end{bmatrix},\quad B=\begin{bmatrix}b_1&0&\cdots&0\\0&b_2&\cdots&0\\\vdots&\vdots&\ddots&\vdots\\0&0&\cdots&b_n\end{bmatrix}.
\]

\begin{enumerate}[label=\alph*)]

\item Compute $AB$.

\begin{answer}
If $A=\operatorname{diag}(a_1,\ldots,a_n)$ and
$B=\operatorname{diag}(b_1,\ldots,b_n)$, then
$(AB)_{ij}=\sum_k a_{ik}b_{kj}$ vanishes when $i\ne j$ and equals $a_ib_i$
when $i=j$. Thus $AB=\operatorname{diag}(a_1b_1,\ldots,a_nb_n)$.
\end{answer}

\item What conditions must $a_1,a_2,\ldots,a_n$ satisfy for $A$ to be invertible? When $A$ is invertible, what is $A^{-1}$?

\begin{answer}
$A$ is invertible exactly when every $a_j\ne0$, and then
\[
 A^{-1}=\operatorname{diag}(a_1^{-1},\ldots,a_n^{-1}).
\]
Both products with $A$ equal $I_n$. If $a_j=0$, row $j$ of $AX$ is zero for
every $X$, so $AX$ cannot equal $I_n$.
\end{answer}
\end{enumerate}
