$C=
\begin{bmatrix}
    0 & 0 & 3 & 3i & -15 \\
    i & 3 & 2 & 0 & 3 \\
    2 & -6i & -5i & 1 & 2
\end{bmatrix}
$ \\[1.0em]

\begin{enumerate}
    \item
    \textbf{Row Echelon:} \\[1.0em]
    \begin{answer}
        $C \xrightarrow[t_1]{R_1 \leftrightarrow R_2}
        \begin{bmatrix}
            i & 3 & 2 & 0 & 3 \\
            0 & 0 & 3 & 3i & -15 \\
            2 & -6i & -5i & 1 & 2
        \end{bmatrix}
        $ \\[1.0em]

        $\xrightarrow[t_2]{
            \substack{
                R_1 \leftarrow -iR_1 \\
                R_2 \leftarrow \frac{1}{3}R_2 \\
                R_3 \leftarrow R_3 - 2R_1
            }
        }
        \begin{bmatrix}
            1 & -3i & -2i & 0 & -3i \\
            0 & 0 & 1 & i & -5 \\
            0 & 0 & -i & 1 & 2+6i
        \end{bmatrix}
        $
        $\xrightarrow[t_3]{R_3 \leftarrow R_3+iR_2}
        \begin{bmatrix}
            1 & -3i & -2i & 0 & -3i \\
            0 & 0 & 1 & i & -5 \\
            0 & 0 & 0 & 0 & 2+i
        \end{bmatrix}
        = C'$
    \end{answer}

    \item
    \textbf{Rank:}
    \begin{answer}
        $\operatorname{rank}(C)=3.$
    \end{answer} \\

    \item
    \textbf{Reduced Echelon form:} \\[1.0em]
    \begin{answer}
        $C' \xrightarrow[t_4]{R_3 \leftarrow \frac{1}{2+i}R_3}
        \begin{bmatrix}
            1 & -3i & -2i & 0 & -3i \\
            0 & 0 & 1 & i & -5 \\
            0 & 0 & 0 & 0 & 1
        \end{bmatrix}
        $ \\[1.0em]

        $\xrightarrow[t_5]{
            \substack{
                R_2 \leftarrow R_2+5R_3 \\
                R_1 \leftarrow R_1+3iR_3 \\
                R_1 \leftarrow R_1+2iR_2
            }
        }
        \begin{bmatrix}
            1 & -3i & 0 & -2 & 0 \\
            0 & 0 & 1 & i & 0 \\
            0 & 0 & 0 & 0 & 1
        \end{bmatrix}
        = \operatorname{rref}(C)$
    \end{answer}
\end{enumerate}
