\setcounter{enumi}{18}
\item {\bfseries\boldmath
Discuss the rank of each of the following matrices in terms of the real parameters $\alpha$ and $\beta$ occurring in it.
}

\begin{enumerate}[label=\alph*)]

\item $A=\begin{bmatrix}-5&1&8\\0&\alpha&0\\0&0&\alpha-1\end{bmatrix}$

\begin{answer}
$A=\begin{bmatrix}-5&1&8\\0&\alpha&0\\0&0&\alpha-1\end{bmatrix}$.
If $\alpha\notin\{0,1\}$, there are three pivots. For $\alpha=0$ or $1$,
exactly two rows remain independent. Thus
\[
 \rank(A)=\begin{cases}2,&\alpha\in\{0,1\},\\3,&\alpha\notin\{0,1\}.\end{cases}
\]
\end{answer}

\item $B=\begin{bmatrix}4&\alpha-1&3\\0&\alpha^2+1&\alpha-1\\0&0&\alpha+5\end{bmatrix}$

\begin{answer}
$B=\begin{bmatrix}4&\alpha-1&3\\0&\alpha^2+1&\alpha-1\\0&0&\alpha+5\end{bmatrix}$.
Since $4\ne0$ and $\alpha^2+1>0$, the first two pivots always exist.
Therefore $\rank(B)=2$ for $\alpha=-5$, and $\rank(B)=3$ otherwise.
\end{answer}

\item $C=\begin{bmatrix}-1&1&3\\1&\alpha-1&-\alpha-3\\-1-\alpha&2-\alpha&3\alpha+5\end{bmatrix}$

\begin{answer}
For
$C=\begin{bmatrix}-1&1&3\\1&\alpha-1&-\alpha-3\\-1-\alpha&2-\alpha&3\alpha+5\end{bmatrix}$,
apply $F_2\gets F_2+F_1$, $F_3\gets F_3-(1+\alpha)F_1$,
$F_3\gets F_3+2F_2$, $F_2\leftrightarrow F_3$,
$F_3\gets F_3-\alpha F_2$. This gives
\[
 C\sim\begin{bmatrix}-1&1&3\\0&1&2-2\alpha\\0&0&\alpha(2\alpha-3)\end{bmatrix}.
\]
No division by a parameter was used. Hence $\rank(C)=2$ for
$\alpha\in\{0,\frac32\}$, and $\rank(C)=3$ otherwise.
\end{answer}

\item $D=\begin{bmatrix}1&2&-1&2\\3&2\alpha+6&\beta^2-4&\beta^2+9\\-1&-2&\beta^2&\beta-1\end{bmatrix}$

\begin{answer}
For
\[
D=\begin{bmatrix}1&2&-1&2\\3&2\alpha+6&\beta^2-4&\beta^2+9\\-1&-2&\beta^2&\beta-1\end{bmatrix},
\]
$F_2\gets F_2-3F_1$, $F_3\gets F_3+F_1$ give
\[
 D\sim\begin{bmatrix}1&2&-1&2\\0&2\alpha&\beta^2-1&\beta^2+3\\0&0&\beta^2-1&\beta+1\end{bmatrix}.
\]
If $\beta=-1$, the third row is zero and the second is nonzero.
If $\beta=1$, the last two rows are independent exactly when $\alpha\ne0$.
If $\beta\notin\{-1,1\}$ and $\alpha\ne0$, there are three pivots.
If $\beta\notin\{-1,1\}$ and $\alpha=0$, subtracting row $3$ from row $2$
produces the nonzero last entry
$\beta^2-\beta+2=(\beta-\frac12)^2+\frac74$;
the last two rows are again independent. Therefore
\[
 \rank(D)=\begin{cases}
 2,&\beta=-1\text{, or }(\beta=1\text{ and }\alpha=0),\\
 3,&\text{otherwise}.
 \end{cases}
\]
\end{answer}
\end{enumerate}
