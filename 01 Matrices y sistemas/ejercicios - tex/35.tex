\item Let $a,b\in\mathbb R$ and consider
the following system of linear equations
in the unknowns $x$, $y$, $z$:

$$
(S_{a,b}):
\begin{cases}
-x+ay+bz=b\\
x-y+z=2\\
x-ay+z=1
\end{cases}
$$

\begin{enumerate}[label=\alph*)]

\item Discuss the system $S_{a,b}$
depending on the parameters $a$ and $b$.

\begin{answer}
    The augmented matrix is
    $$
    \left[
    \begin{array}{ccc|c}
    -1&a&b&b\\
    1&-1&1&2\\
    1&-a&1&1
    \end{array}
    \right].
    $$

    $$
    \xrightarrow{R_1\leftrightarrow R_3}
    \left[
    \begin{array}{ccc|c}
    1&-a&1&1\\
    1&-1&1&2\\
    -1&a&b&b
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_2\leftarrow R_2-R_1}
    \left[
    \begin{array}{ccc|c}
    1&-a&1&1\\
    0&a-1&0&1\\
    -1&a&b&b
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_3\leftarrow R_3+R_1}
    \left[
    \begin{array}{ccc|c}
    1&-a&1&1\\
    0&a-1&0&1\\
    0&0&b+1&b+1
    \end{array}
    \right].
    $$

    {\color{red}
    \textbf{Note:} The values $a=1$
    and $b=-1$ must be examined
    separately, since they cause
    potential pivots to vanish.
    }

    \textbf{Case 1: $a=1$, $b\neq-1$}

    $$
    \left[
    \begin{array}{ccc|c}
    1&-1&1&1\\
    0&0&0&1\\
    0&0&b+1&b+1
    \end{array}
    \right]
    $$

    Since $b+1\neq0$,
    $$
    r(A)=2<3=r([A\mid B]).
    $$

    The system is inconsistent.

    \textbf{Case 2: $a=1$, $b=-1$}

    $$
    \left[
    \begin{array}{ccc|c}
    1&-1&1&1\\
    0&0&0&1\\
    0&0&0&0
    \end{array}
    \right]
    $$

    $$
    r(A)=1<2=r([A\mid B]).
    $$

    The system is inconsistent.

    \textbf{Case 3: $a\neq1$, $b=-1$}

    $$
    \left[
    \begin{array}{ccc|c}
    1&-a&1&1\\
    0&a-1&0&1\\
    0&0&0&0
    \end{array}
    \right]
    $$

    $$
    r(A)=r([A\mid B])=2<3.
    $$

    The system is consistent and
    indeterminate, with degree
    $$
    p=3-2=1.
    $$

    \textbf{Case 4: $a\neq1$, $b\neq-1$}

    All three pivot positions
    are nonzero. Hence
    $$
    r(A)=r([A\mid B])=3=n.
    $$

    The system is consistent
    and determined.

    \textbf{Conclusion}

    $$
    \boxed{
    \begin{array}{ll}
    a=1:
    &\text{inconsistent},\\
    a\neq1,\ b=-1:
    &\text{consistent indeterminate, degree }1,\\
    a\neq1,\ b\neq-1:
    &\text{consistent determined}.
    \end{array}
    }
    $$
\end{answer}

\item Determine the values of $a$
and $b$ for which the associated
homogeneous system has a nonzero solution.

\begin{answer}
    The associated homogeneous system has the same coefficient matrix $A$ and zero independent terms.

    From the row operations in part a), its augmented matrix is equivalent to
    $$
    \left[
    \begin{array}{ccc|c}
    1&-a&1&0\\
    0&a-1&0&0\\
    0&0&b+1&0
    \end{array}
    \right].
    $$

    Every homogeneous system admits the trivial solution $X=0$.

    A nonzero solution exists if and only if
    $$
    r(A)<n=3.
    $$

    This happens when at least one of the last two potential pivots vanishes:
    $$
    a-1=0
    \quad\text{or}\quad
    b+1=0.
    $$

    Therefore,
    $$
    \boxed{a=1\quad\text{or}\quad b=-1}.
    $$

    More precisely,
    $$
    \begin{array}{c|c|c}
    \text{Conditions}&r(A)&
    \text{Degree}\\
    \hline
    a\neq1,\ b\neq-1&3&0\\
    a=1,\ b\neq-1&2&1\\
    a\neq1,\ b=-1&2&1\\
    a=1,\ b=-1&1&2
    \end{array}
    $$

    {\color{red}
    \textbf{Note:} The homogeneous system is always consistent,
    even if the original system is inconsistent.
    It admits nonzero solutions precisely when $r(A)<n$.}
\end{answer}

\item Determine the values of $a$
and $b$ for which the coefficient
matrix $A$ is invertible.

\begin{answer}
    Since $A$ is a square matrix
    of order $3$,
    $$
    A\text{ invertible}
    \iff r(A)=3.
    $$

    From part a), the potential
    pivot positions are
    $$
    1,\qquad a-1,\qquad b+1.
    $$

    All three are nonzero exactly when
    $$
    a\neq1
    \quad\text{and}\quad
    b\neq-1.
    $$

    Hence
    $$
    \boxed{a\neq1,\qquad b\neq-1}.
    $$

    {\color{red}
    \textbf{Note:} The inverse does not
    need to be calculated. For a square
    matrix of order $n$, full rank
    is equivalent to invertibility.
    }
\end{answer}

\item Solve the system $S_{-3,-1}$
using Gauss-Jordan elimination.

\begin{answer}
    Substituting $a=-3$ and $b=-1$
    gives
    $$
    \begin{cases}
    -x-3y-z=-1\\
    x-y+z=2\\
    x+3y+z=1
    \end{cases}
    $$

    Its augmented matrix is
    $$
    \left[
    \begin{array}{ccc|c}
    -1&-3&-1&-1\\
    1&-1&1&2\\
    1&3&1&1
    \end{array}
    \right].
    $$

    $$
    \xrightarrow{R_1\leftrightarrow R_3}
    \left[
    \begin{array}{ccc|c}
    1&3&1&1\\
    1&-1&1&2\\
    -1&-3&-1&-1
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_2\leftarrow R_2-R_1}
    \left[
    \begin{array}{ccc|c}
    1&3&1&1\\
    0&-4&0&1\\
    -1&-3&-1&-1
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_3\leftarrow R_3+R_1}
    \left[
    \begin{array}{ccc|c}
    1&3&1&1\\
    0&-4&0&1\\
    0&0&0&0
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_2\leftarrow-\frac14 R_2}
    \left[
    \begin{array}{ccc|c}
    1&3&1&1\\
    0&1&0&-\frac14\\
    0&0&0&0
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_1\leftarrow R_1-3R_2}
    \left[
    \begin{array}{ccc|c}
    1&0&1&\frac74\\
    0&1&0&-\frac14\\
    0&0&0&0
    \end{array}
    \right].
    $$

    Therefore,
    $$
    r(A)=r([A\mid B])=2<3=n.
    $$

    The system is consistent and
    indeterminate, with degree
    $$
    p=3-2=1.
    $$

    {\color{red}
    \textbf{Note:} The pivot columns
    are $1$ and $2$. Therefore $z$
    is the free variable.
    }

    From the reduced matrix,
    $$
    \begin{cases}
    x+z=\frac74\\
    y=-\frac14
    \end{cases}
    \iff
    \begin{cases}
    x=\frac74-z\\
    y=-\frac14
    \end{cases}
    $$

    Set
    $$
    z=\alpha,\qquad\alpha\in\mathbb R.
    $$

    Thus,
    $$
    \begin{cases}
    x=\frac74-\alpha\\
    y=-\frac14\\
    z=\alpha
    \end{cases}
    $$

    \textbf{General solution}

    $$
    \begin{aligned}
    (x,y,z)
    &=\left(\frac74-\alpha,
                -\frac14,\alpha\right)\\
    &=\left(\frac74,-\frac14,0\right)
        +(-\alpha,0,\alpha)\\
    &=\left(\frac74,-\frac14,0\right)
        +\alpha(-1,0,1).
    \end{aligned}
    $$

    Therefore,
    $$
    \boxed{
    (x,y,z)=
    \left(\frac74,-\frac14,0\right)
    +\alpha(-1,0,1),
    \quad\alpha\in\mathbb R.
    }
    $$

    The general solution of the associated homogeneous system is
    $$
    \boxed{
    (x,y,z)=\alpha(-1,0,1),
    \quad\alpha\in\mathbb R.
    }
    $$
\end{answer}

\end{enumerate}
