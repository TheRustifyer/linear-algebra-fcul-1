$I=
\begin{bmatrix}
    5
\end{bmatrix}
$ \\[1.0em]

\begin{enumerate}
    \item
    \textbf{Row Echelon:}
    \begin{answer}
        $I$ is already in row echelon form.
    \end{answer}

    \item
    \textbf{Rank:}
    \begin{answer}
        $\operatorname{rank}(I)=1.$
    \end{answer} \\

    \item
    \textbf{Reduced Echelon form:} \\[1.0em]
    \begin{answer}
        $I \xrightarrow[t_1]{R_1 \leftarrow \frac{1}{5}R_1}
        \begin{bmatrix}
            1
        \end{bmatrix}
        = \operatorname{rref}(I)$
    \end{answer}
\end{enumerate}
