
\item \textbf{Discuss the following systems
of linear equations in the unknowns
$x$, $y$, $z$, depending on the parameter
$a\in\mathbb{R}$. For indeterminate
systems, determine the degree of
indeterminacy}

\begin{enumerate}[label=\alph*)]

    \item
    $$
    \begin{cases}
        x+y-az=0\\
        ax+y-z=1-a\\
        x+ay-z=1-a
    \end{cases}
    $$

    $$
    \begin{cases}
        x+y-z=0\\
        ay+(2a-1)z=a\\
        ax+2ay=0\\
        ax+ay+(1-2a)z=a-1
    \end{cases}
    $$

\end{enumerate}
