\setcounter{enumi}{21}
\item {\bfseries\boldmath
Let $A_1,A_2,\ldots,A_n$ be invertible square matrices of the same order. Express $(A_1A_2\cdots A_n)^{-1}$ in terms of $A_1^{-1},A_2^{-1},\ldots,A_n^{-1}$.
}

\begin{answer}
\[
 (A_1A_2\cdots A_n)^{-1}=A_n^{-1}A_{n-1}^{-1}\cdots A_1^{-1}.
\]
In either order of multiplication, associativity allows adjacent factors
$A_jA_j^{-1}$ or $A_j^{-1}A_j$ to cancel successively, leaving the identity.
\end{answer}
