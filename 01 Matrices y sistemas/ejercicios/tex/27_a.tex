\begin{answer}
    {\color{red}
        \textbf{Note:} To determine whether $A_\alpha$ is invertible, it is sufficient to obtain
        a row echelon form and verify that $\operatorname{rank}(A_\alpha)=4$.
    }

    $$
    A_\alpha=
    \begin{bmatrix}
        1 & \alpha & \alpha & \alpha\\
        0 & 0 & \alpha+3 & \alpha+2\\
        1 & \alpha & \alpha & -1\\
        -1 & 1-\alpha & 1 & 7
    \end{bmatrix}
    \xrightarrow{R_4\leftarrow R_4+R_3}
    \begin{bmatrix}
        1 & \alpha & \alpha & \alpha\\
        0 & 0 & \alpha+3 & \alpha+2\\
        1 & \alpha & \alpha & -1\\
        0 & 1 & \alpha+1 & 6
    \end{bmatrix}
    $$

    $$
    \xrightarrow{R_2\leftrightarrow R_4}
    \begin{bmatrix}
        1 & \alpha & \alpha & \alpha\\
        0 & 1 & \alpha+1 & 6\\
        1 & \alpha & \alpha & -1\\
        0 & 0 & \alpha+3 & \alpha+2
    \end{bmatrix}
    \xrightarrow{R_3\leftarrow R_3-R_1}
    \begin{bmatrix}
        1 & \alpha & \alpha & \alpha\\
        0 & 1 & \alpha+1 & 6\\
        0 & 0 & 0 & -\alpha-1\\
        0 & 0 & \alpha+3 & \alpha+2
    \end{bmatrix}
    $$

    $$
    \xrightarrow{R_3\leftrightarrow R_4}
    \begin{bmatrix}
        1 & \alpha & \alpha & \alpha\\
        0 & 1 & \alpha+1 & 6\\
        0 & 0 & \alpha+3 & \alpha+2\\
        0 & 0 & 0 & -\alpha-1
    \end{bmatrix}
    $$

    \textbf{Case 1: $\alpha\notin\{-3,-1\}$}

    $$
    \alpha+3\neq0
    \land
    -\alpha-1\neq0
    \implies
    \operatorname{rank}(A_\alpha)=4
    $$

    \textbf{Case 2: $\alpha=-3$}

    $$
    \begin{bmatrix}
        1 & -3 & -3 & -3\\
        0 & 1 & -2 & 6\\
        0 & 0 & 0 & -1\\
        0 & 0 & 0 & 2
    \end{bmatrix}
    \xrightarrow{R_4\leftarrow R_4+2R_3}
    \begin{bmatrix}
        1 & -3 & -3 & -3\\
        0 & 1 & -2 & 6\\
        0 & 0 & 0 & -1\\
        0 & 0 & 0 & 0
    \end{bmatrix}
    $$

    $$
    \operatorname{rank}(A_{-3})=3<4
    $$

    \textbf{Case 3: $\alpha=-1$}

    $$
    \begin{bmatrix}
        1 & -1 & -1 & -1\\
        0 & 1 & 0 & 6\\
        0 & 0 & 2 & 1\\
        0 & 0 & 0 & 0
    \end{bmatrix}
    $$

    $$
    \operatorname{rank}(A_{-1})=3<4
    $$

    $$
    \boxed{
        A_\alpha\text{ invertible}
        \iff
        \alpha\in\mathbb{R}\setminus\{-3,-1\}
    }
    $$
\end{answer}
