\item \textbf{Let $A\in M_n(K)$ be an invertible matrix. Prove that:}

\begin{enumerate}[label=\alph*)]

\item \textbf{If $B\in M_{n\times m}(K)$
satisfies $AB=0_{n\times m}$, then
$B=0_{n\times m}$.}

\begin{answer}
    Since $A$ is invertible, multiplying
    both sides by $A^{-1}$ on the left,

    $$
    \begin{aligned}
    AB &= 0_{n\times m}\\
    A^{-1}(AB) &= A^{-1}0_{n\times m}\\
    (A^{-1}A)B &= 0_{n\times m}\\
    I_nB &= 0_{n\times m}\\
    B &= 0_{n\times m}.
    \end{aligned}
    $$
\end{answer}

\item \textbf{If $B\in M_{m\times n}(K)$
satisfies $BA=0_{m\times n}$, then
$B=0_{m\times n}$.}

\begin{answer}
    Multiplying both sides by $A^{-1}$
    on the right,

    $$
    \begin{aligned}
    BA &= 0_{m\times n}\\
    (BA)A^{-1} &= 0_{m\times n}A^{-1}\\
    B(AA^{-1}) &= 0_{m\times n}\\
    BI_n &= 0_{m\times n}\\
    B &= 0_{m\times n}.
    \end{aligned}
    $$
\end{answer}

\item \textbf{If $B,C\in M_{n\times m}(K)$
satisfy $AB=AC$, then $B=C$.}

\begin{answer}
    Multiplying both sides by $A^{-1}$
    on the left,

    $$
    \begin{aligned}
    AB &= AC\\
    A^{-1}(AB) &= A^{-1}(AC)\\
    (A^{-1}A)B &= (A^{-1}A)C\\
    I_nB &= I_nC\\
    B &= C.
    \end{aligned}
    $$
\end{answer}

\item \textbf{If $B,C\in M_{m\times n}(K)$
satisfy $BA=CA$, then $B=C$.}

\begin{answer}
    Multiplying both sides by $A^{-1}$
    on the right,

    $$
    \begin{aligned}
    BA &= CA\\
    (BA)A^{-1} &= (CA)A^{-1}\\
    B(AA^{-1}) &= C(AA^{-1})\\
    BI_n &= CI_n\\
    B &= C.
    \end{aligned}
    $$
\end{answer}

\end{enumerate}
