
\begin{answer}
    The augmented matrix is
    $$
    \left[
    \begin{array}{ccc|c}
    1&1&-1&0\\
    0&a&2a-1&a\\
    a&2a&0&0\\
    a&a&1-2a&a-1
    \end{array}
    \right].
    $$

    $$
    \xrightarrow{R_3\leftarrow R_3-aR_1}
    \left[
    \begin{array}{ccc|c}
    1&1&-1&0\\
    0&a&2a-1&a\\
    0&a&a&0\\
    a&a&1-2a&a-1
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_4\leftarrow R_4-aR_1}
    \left[
    \begin{array}{ccc|c}
    1&1&-1&0\\
    0&a&2a-1&a\\
    0&a&a&0\\
    0&0&1-a&a-1
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_3\leftarrow R_3-R_2}
    \left[
    \begin{array}{ccc|c}
    1&1&-1&0\\
    0&a&2a-1&a\\
    0&0&1-a&-a\\
    0&0&1-a&a-1
    \end{array}
    \right]
    $$

    $$
    \xrightarrow{R_4\leftarrow R_4-R_3}
    \left[
    \begin{array}{ccc|c}
    1&1&-1&0\\
    0&a&2a-1&a\\
    0&0&1-a&-a\\
    0&0&0&2a-1
    \end{array}
    \right].
    $$

    {\color{red}
    \textbf{Note:} The last row represents
    $0=2a-1$. Hence compatibility is possible only when $a=\frac12$.}

    \textbf{Case 1: $a=\frac12$}

    $$
    \left[
    \begin{array}{ccc|c}
    1&1&-1&0\\
    0&\frac12&0&\frac12\\
    0&0&\frac12&-\frac12\\
    0&0&0&0
    \end{array}
    \right]
    $$

    $$
    r(A)=r([A\mid B])=3=n.
    $$

    The system is consistent and
    determined.

    In particular,
    $$
    \begin{cases}
    y=1\\
    z=-1\\
    x+y-z=0
    \end{cases}
    \implies
    (x,y,z)=(-2,1,-1).
    $$

    \textbf{Case 2: $a=0$ or $a=1$}

    $$
    r(A)=2<3=r([A\mid B]).
    $$

    The system is inconsistent.

    \textbf{Case 3: $a\neq0,\frac12,1$}

    $$
    r(A)=3<4=r([A\mid B]).
    $$

    The system is inconsistent.

    \textbf{Conclusion}

    $$
    \boxed{
    \begin{cases}
    a=\frac12:
    &\text{consistent and determined},\\
    a\neq\frac12:
    &\text{inconsistent}.
    \end{cases}
    }
    $$

    There are no consistent indeterminate cases.
\end{answer}
